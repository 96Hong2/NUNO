# D5: 임시 시험 파일 셋(.only, .skip, 단언 없음)을 하나씩 두고 pnpm lint
# 날짜: 2026-11-24 06:22 KST, 커밋 bae1901

$ cat packages/twin/src/domain/probe-only.test.ts
import { expect, it } from "vitest";

it.only("집중 시험", () => {
  expect(1).toBe(1);
});
$ pnpm lint
$ eslint --cache --cache-location node_modules/.cache/eslint/ --max-warnings 0 .
